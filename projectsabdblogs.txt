void setup() {
  pinMode(leftIR, INPUT);

void loop() {
  bool leftState = digitalRead(leftIR);   // HIGH on black line

C++
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
const int trigPin = 9;
const int echoPin = 10;
const int buzzerPin = 11;

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(buzzerPin, OUTPUT);
  lcd.init();
  lcd.backlight();
}

void loop() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH);
  float distance = duration * 0.0343 / 2;

  lcd.setCursor(0, 0);
  lcd.print("Distance Metric:");
  lcd.setCursor(0, 1);
  lcd.print(distance);
  lcd.print(" cm    ");

  if (distance > 0 && distance < 10) {
    digitalWrite(buzzerPin, HIGH);
  } else {
    digitalWrite(buzzerPin, LOW);
  }
  delay(100);
}

Introduction:

Why reliable soldering is critical for clean hardware prototyping.

Essential Budget Tools:

60W Adjustable Temperature Soldering Iron (PKR 1,200 - 2,000).

Rosin-core 60/40 solder wire, soldering paste/flux, brass wire tip cleaner, heat shrink sleeves.

Step-by-Step Soldering Method:

Step 1: Tinning the soldering tip with fresh solder.

Step 2: Heating both component leg and PCB pad simultaneously (2-3 seconds).

Step 3: Feeding solder wire directly into the joint.

Step 4: Removing wire first, then the iron, allowing natural cooling.

Identifying Joints:

Good Joint: Shiny, concave cone shape ("volcano").

Cold Joint: Dull, grainy, non-conductive blob.

Solder Bridge: Accidental short circuit across adjacent pads.

Safety First:

Fume ventilation, eye protection, avoiding lead exposure.
